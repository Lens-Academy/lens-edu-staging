      bb[i].classList.toggle("is-active", key === state.band);

      bb[i].classList.toggle("is-seen", state.seen.indexOf(key) !== -1);
    }
    var rb = regionsEl.querySelectorAll("button");
    for (i = 0; i < rb.length; i++) {
      var on = rb[i].dataset.region === state.region || (state.region === null && rb[i].dataset.region === "all");
      rb[i].classList.toggle("is-active", on);
      rb[i].setAttribute("aria-pressed", on ? "true" : "false");
    }
    for (var k in bandGroups) if (Object.prototype.hasOwnProperty.call(bandGroups, k)) {
      bandGroups[k].classList.toggle("is-active", k === state.band);
      bandGroups[k].classList.toggle("is-dim", state.band !== null && k !== state.band);
    }
    var cells = chart.querySelectorAll(".cell");
    var want = state.region === null ? -1 : (state.region === "us" ? 0 : state.region === "cn" ? 1 : 2);
    for (i = 0; i < cells.length; i++) {
      cells[i].classList.toggle("is-dim", want !== -1 && !cells[i].classList.contains("r" + want));
    }
    renderDetail();
    var left = DC_BANDS.filter(function (k) { return state.seen.indexOf(k) === -1; }).length;
    statusEl.textContent = isDone()
      ? "Done: you have opened all six datacenter size bands."
      : "Opened " + (DC_BANDS.length - left) + " of the 6 datacenter size bands.";
    statusEl.classList.toggle("is-done", isDone());
  }

    if (b) {
      s += "Currently open: " + b.label + " (" + b.sub + "), verification measure " + b.measure + ".";
      if (b.counts[0] + b.counts[1] + b.counts[2] > 0) s += " Drawn datacenters: US " + b.counts[0] + ", China " + b.counts[1] + ", rest of world " + b.counts[2] + ".";
      s += " ";
    } else {
      s += "No band open yet. ";
    }
    s += "Region filter: " + (state.region === null ? "all regions" : (state.region === "us" ? "US" : state.region === "cn" ? "China" : "rest of world")) + ". ";
    s += "Bands opened so far: " + (state.seen.length ? state.seen.map(function (k) { return bandByKey(k).label; }).join(", ") : "none") + ".";
    return s;
  }

      if (isDone() && !completed) { completed = true; Lens.complete(); }
    } else {
      try { localStorage.setItem("ai-2040-compute-locations", JSON.stringify(json)); } catch (e) {}
    }
  }

  function hydrate(saved, meta) {
    if (saved && typeof saved === "object") {
      if (typeof saved.band === "string" && bandByKey(saved.band)) state.band = saved.band;
      if (saved.region === "us" || saved.region === "cn" || saved.region === "row") state.region = saved.region;
      if (Array.isArray(saved.seen)) state.seen = saved.seen.filter(function (k) { return !!bandByKey(k); });
    }
    completed = !!(meta && meta.completed);
    if (completed) state.seen = DC_BANDS.slice();
    render();
  }

  render();
  if (window.Lens) {
    Lens.onState(hydrate);
  } else {
    try { var raw = localStorage.getItem("ai-2040-compute-locations"); if (raw) hydrate(JSON.parse(raw), null); } catch (e) {}
  }
